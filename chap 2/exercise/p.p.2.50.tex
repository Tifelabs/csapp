\begin{flushleft}
Show how the following binary fractional values would be rounded to the nearest half
($1$ bit to the right of the binary point), according to the round-to-even rule.
In each case, show the numeric values, both before and after the rounding.

\bigskip
\textbf{Solution}

\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Original} & \textbf{Rounded} \\
\midrule
    A.\ $10.111_2 = 2.875_{10}$ & $11.0_2 = 3.0_{10}$ \\
    B.\ $11.010_2 = 3.25_{10}$  & $11.0_2 = 3.0_{10}$ \\
    C.\ $11.000_2 = 3.0_{10}$   & $11.0_2 = 3.0_{10}$ \\
    D.\ $10.110_2 = 2.75_{10}$  & $11.0_2 = 3.0_{10}$ \\
\bottomrule
\end{tabular}

\bigskip
\textbf{Explanation}

\begin{itemize}
  \item \textbf{A.} $10.111_2$ \\
  Bits after the rounding position: $11_2 > 1000\ldots_2$ $\Rightarrow$ round up to $11.0_2$.

  \item \textbf{B.} $11.010_2$ \\
  Exactly halfway between $11.0_2$ and $11.1_2$. \\
  Round-to-even selects the even LSB $\Rightarrow$ $11.0_2$.

  \item \textbf{C.} $11.000_2$ \\
  Already exact at the required precision $\Rightarrow$ remains $11.0_2$.

  \item \textbf{D.} $10.110_2$ \\
  Exactly halfway between $10.1_2$ and $11.0_2$. \\
  Round-to-even selects the even LSB $\Rightarrow$ $11.0_2$.
\end{itemize}
\end{flushleft}
